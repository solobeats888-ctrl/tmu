name: Build macOS

on:
  push:
  workflow_dispatch:

jobs:
  build:
    runs-on: macos-14
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4

      - name: Compilar (Apple Silicon + Intel)
        run: |
          cmake -B build -G Xcode -DCMAKE_OSX_ARCHITECTURES="arm64;x86_64" -DCMAKE_OSX_DEPLOYMENT_TARGET=10.13
          cmake --build build --config Release -- -quiet

      - name: Firmar (ad-hoc)
        run: |
          A=build/TMU_artefacts/Release
          codesign --force --deep --sign - "$A/VST3/TMU.vst3"
          codesign --force --deep --sign - "$A/AU/TMU.component"
          lipo -info "$A/VST3/TMU.vst3/Contents/MacOS/TMU"

      - name: Paquetes del plugin
        run: |
          set -e
          A=build/TMU_artefacts/Release
          mkdir -p out stage/vst3 stage/au
          ditto "$A/VST3/TMU.vst3" stage/vst3/TMU.vst3
          ditto "$A/AU/TMU.component" stage/au/TMU.component
          chmod +x installer/scripts-vst3/* installer/scripts-au/*
          for b in vst3:TMU.vst3 au:TMU.component; do
            kind=${b%%:*}; bundle=${b#*:}
            printf '%s\n' '<?xml version="1.0" encoding="UTF-8"?>' \
              '<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">' \
              '<plist version="1.0"><array><dict>' \
              '<key>BundleHasStrictIdentifier</key><false/>' \
              '<key>BundleIsRelocatable</key><false/>' \
              '<key>BundleIsVersionChecked</key><false/>' \
              '<key>BundleOverwriteAction</key><string>upgrade</string>' \
              "<key>RootRelativeBundlePath</key><string>$bundle</string>" \
              '</dict></array></plist>' > comp-$kind.plist
            plutil -lint comp-$kind.plist
          done
          pkgbuild --root stage/vst3 --component-plist comp-vst3.plist --identifier com.viszeversa.tmu.vst3 \
            --version 1.0.0 --install-location /Library/Audio/Plug-Ins/VST3 --scripts installer/scripts-vst3 out/TMU-VST3.pkg
          pkgbuild --root stage/au --component-plist comp-au.plist --identifier com.viszeversa.tmu.au \
            --version 1.0.0 --install-location /Library/Audio/Plug-Ins/Components --scripts installer/scripts-au out/TMU-AU.pkg
          productbuild --package out/TMU-VST3.pkg --package out/TMU-AU.pkg out/TMU-Instalador.pkg
          rm out/TMU-VST3.pkg out/TMU-AU.pkg   # queda un solo instalador (solo el plugin, sin soundfonts)

      - name: Prueba del instalador
        run: |
          set -e
          sudo installer -pkg out/TMU-Instalador.pkg -target / -verboseR
          ls -la /Library/Audio/Plug-Ins/VST3/TMU.vst3 /Library/Audio/Plug-Ins/Components/TMU.component
          pkgutil --pkgs | grep viszeversa
          echo "INSTALADOR OK"

      - name: Validar Audio Unit (auval)
        run: |
          killall -9 AudioComponentRegistrar || true
          sleep 3
          auval -a | grep -i viszeversa || true
          auval -strict -v aumu Tmu1 Vszv

      - name: Validar VST3 (pluginval)
        continue-on-error: true
        run: |
          curl -sL -o pluginval.zip https://github.com/Tracktion/pluginval/releases/download/v1.0.4/pluginval_macOS.zip
          unzip -q pluginval.zip
          ./pluginval.app/Contents/MacOS/pluginval --strictness-level 5 --skip-gui-tests --validate /Library/Audio/Plug-Ins/VST3/TMU.vst3

      - name: Subir resultado
        uses: actions/upload-artifact@v4
        with:
          name: TMU-mac
          path: out/
